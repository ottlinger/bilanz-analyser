9   import java.math.BigDecimal;
10  import java.time.LocalDate;
11  import java.util.Optional;

13  @NoArgsConstructor(access = AccessLevel.PRIVATE)
14  @Log4j2
15  public final class BilanzRowParser {
16  
17      public static Optional<BilanzRow> fromOdfTableRow(final OdfTableRow row) {
18          try {
19              BilanzRow bilanzRow = BilanzRow.builder() //
20                      // expected format: yyyy-MM-dd
21                      .date(LocalDate.parse(row.getCellByIndex(0).getStringValue()))
22                      // remove trailing spaces and currency symbol
23                      .amount(new BigDecimal(cleanUpAmount(row.getCellByIndex(1).getStringValue()))) //
24                      .description(row.getCellByIndex(2).getStringValue()) //
25                      .shop(row.getCellByIndex(3).getStringValue()) //
26                      .payment(row.getCellByIndex(4).getStringValue()) //
27                      .category(row.getCellByIndex(5).getStringValue()).build();
28  
29              return Optional.of(bilanzRow);
30          } catch (Exception e) {
31              log.error("Skipping row due to: {}", e.getMessage());
32              return Optional.empty();
33          }
34      }
35  
36      /**
37       * Removes currency symbol, changes <code>,</code> to <code>.</code> and trims the given amount value from an ODS file.
38       *
39       * @param amount given amount, e.g. 1,23 €
40       * @return trimmed amount in order to be parseable as a numeric.
41       */
42      static String cleanUpAmount(String amount) {
43          if (amount != null && !amount.isEmpty()) {
44              return amount.replace("€", "").replace(",", ".").trim();
45          }
46          return amount;
47      }
48  }
